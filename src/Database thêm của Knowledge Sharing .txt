-- =============================================
-- KNOWLEDGE SHARING MODULE FOR ADHD USERS
-- Designed with ADHD-friendly features in mind
-- =============================================

-- Bảng Chủ đề/Danh mục bài viết
CREATE TABLE [dbo].[KnowledgeCategories](
    [Id] [uniqueidentifier] NOT NULL PRIMARY KEY DEFAULT NEWID(),
    [Name] [nvarchar](100) NOT NULL,
    [Slug] [nvarchar](100) NOT NULL UNIQUE,
    [Description] [nvarchar](500) NULL,
    [ColorHex] [nvarchar](7) NOT NULL DEFAULT '#3B82F6',
    [Icon] [nvarchar](50) NULL,
    [ParentCategoryId] [uniqueidentifier] NULL,
    [OrderIndex] [int] NOT NULL DEFAULT 0,
    [IsActive] [bit] NOT NULL DEFAULT 1,
    [CreatedAt] [datetime2](7) NOT NULL DEFAULT GETUTCDATE(),
    CONSTRAINT [FK_KnowledgeCategories_Parent] FOREIGN KEY([ParentCategoryId]) 
        REFERENCES [dbo].[KnowledgeCategories] ([Id])
);

-- Bảng Bài viết
CREATE TABLE [dbo].[Articles](
    [Id] [uniqueidentifier] NOT NULL PRIMARY KEY DEFAULT NEWID(),
    [AuthorId] [uniqueidentifier] NOT NULL,
    [CategoryId] [uniqueidentifier] NULL,
    [Title] [nvarchar](200) NOT NULL,
    [Slug] [nvarchar](250) NOT NULL UNIQUE,
    [Summary] [nvarchar](500) NULL, -- Tóm tắt ngắn gọn cho ADHD
    [Content] [nvarchar](max) NOT NULL,
    [FeaturedImage] [nvarchar](500) NULL,
    [EstimatedReadTime] [int] NULL, -- Thời gian đọc ước tính (phút)
    
    -- ADHD-friendly features
    [HasAudioVersion] [bit] NOT NULL DEFAULT 0, -- Có phiên bản audio không
    [AudioUrl] [nvarchar](500) NULL,
    [HasTLDR] [bit] NOT NULL DEFAULT 0, -- Có phần TL;DR (Too Long Didn't Read)
    [TLDRContent] [nvarchar](1000) NULL,
    [HasVisualAids] [bit] NOT NULL DEFAULT 0, -- Có hình ảnh/infographic
    [DifficultyLevel] [int] NOT NULL DEFAULT 1, -- 1=Dễ, 2=Trung bình, 3=Khó
    
    -- SEO & Metadata
    [MetaDescription] [nvarchar](300) NULL,
    [MetaKeywords] [nvarchar](500) NULL,
    
    -- Publishing
    [Status] [nvarchar](20) NOT NULL DEFAULT 'draft', -- draft, published, archived
    [PublishedAt] [datetime2](7) NULL,
    [ViewCount] [int] NOT NULL DEFAULT 0,
    [LikeCount] [int] NOT NULL DEFAULT 0,
    [CommentCount] [int] NOT NULL DEFAULT 0,
    [ShareCount] [int] NOT NULL DEFAULT 0,
    [BookmarkCount] [int] NOT NULL DEFAULT 0,
    
    -- Timestamps
    [CreatedAt] [datetime2](7) NOT NULL DEFAULT GETUTCDATE(),
    [UpdatedAt] [datetime2](7) NOT NULL DEFAULT GETUTCDATE(),
    [DeletedAt] [datetime2](7) NULL,
    
    CONSTRAINT [FK_Articles_Users] FOREIGN KEY([AuthorId]) 
        REFERENCES [dbo].[Users] ([Id]),
    CONSTRAINT [FK_Articles_Categories] FOREIGN KEY([CategoryId]) 
        REFERENCES [dbo].[KnowledgeCategories] ([Id]) ON DELETE SET NULL
);

-- =============================================
-- TRIGGERS FOR AUTO-UPDATE COUNTS
-- =============================================

-- Trigger để tự động update CommentCount trong Articles
GO
CREATE TRIGGER [trg_Articles_UpdateCommentCount]
ON [dbo].[ArticleComments]
AFTER INSERT, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Update cho INSERT
    UPDATE a
    SET CommentCount = (
        SELECT COUNT(*) 
        FROM ArticleComments 
        WHERE ArticleId = a.Id AND DeletedAt IS NULL
    )
    FROM Articles a
    INNER JOIN inserted i ON a.Id = i.ArticleId;
    
    -- Update cho DELETE
    UPDATE a
    SET CommentCount = (
        SELECT COUNT(*) 
        FROM ArticleComments 
        WHERE ArticleId = a.Id AND DeletedAt IS NULL
    )
    FROM Articles a
    INNER JOIN deleted d ON a.Id = d.ArticleId;
END;
GO

-- Trigger để tự động update LikeCount trong Articles
GO
CREATE TRIGGER [trg_Articles_UpdateLikeCount]
ON [dbo].[ArticleLikes]
AFTER INSERT, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    
    UPDATE a
    SET LikeCount = (SELECT COUNT(*) FROM ArticleLikes WHERE ArticleId = a.Id)
    FROM Articles a
    INNER JOIN (
        SELECT COALESCE(i.ArticleId, d.ArticleId) AS ArticleId
        FROM inserted i FULL OUTER JOIN deleted d ON 1=1
    ) changes ON a.Id = changes.ArticleId;
END;
GO

-- Trigger để tự động update BookmarkCount trong Articles
GO
CREATE TRIGGER [trg_Articles_UpdateBookmarkCount]
ON [dbo].[ArticleBookmarks]
AFTER INSERT, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    
    UPDATE a
    SET BookmarkCount = (SELECT COUNT(*) FROM ArticleBookmarks WHERE ArticleId = a.Id)
    FROM Articles a
    INNER JOIN (
        SELECT COALESCE(i.ArticleId, d.ArticleId) AS ArticleId
        FROM inserted i FULL OUTER JOIN deleted d ON 1=1
    ) changes ON a.Id = changes.ArticleId;
END;
GO
